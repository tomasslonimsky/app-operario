INSTRUCCIONES
1. Abrí sheets.google.com y creá una planilla nueva.
2. Creá una hoja llamada "Stock" con 2 columnas: TipoId y Metros. Completá el stock inicial de cada film (t1, t2, t3, t4, t5, t6).
3. Extensiones → Apps Script. Borrá todo y pegá este código:
function doPost(e){ var ss=SpreadsheetApp.getActiveSpreadsheet(); var d=JSON.parse(e.postData.contents); var sh=ss.getSheetByName(d.sheet)||ss.insertSheet(d.sheet); if(d.action==='append'){ if(sh.getLastRow()===0)sh.appendRow(d.headers); sh.appendRow(d.row); } return ContentService.createTextOutput('ok'); } function doGet(e){ var ss=SpreadsheetApp.getActiveSpreadsheet(); var r={cortes:[],errores:[],stock:{}}; var cs=ss.getSheetByName('Cortes'); if(cs&&cs.getLastRow()>1){ var d=cs.getDataRange().getValues(); for(var i=1;i<d.length;i++){ r.cortes.push({fecha:d[i][0],hora:d[i][1], op:d[i][2],tipo:d[i][3],metros:Number(d[i][4]), comentario:d[i][5]||'',tipoId:d[i][6]||''}); } } var es=ss.getSheetByName('Errores'); if(es&&es.getLastRow()>1){ var d=es.getDataRange().getValues(); for(var i=1;i<d.length;i++){ r.errores.push({fecha:d[i][0],hora:d[i][1], op:d[i][2],tipo:d[i][3],desc:d[i][4]||'', zonas:String(d[i][5]||'')}); } } var st=ss.getSheetByName('Stock'); if(st&&st.getLastRow()>1){ var d=st.getDataRange().getValues(); for(var i=1;i<d.length;i++){ r.stock[String(d[i][0])]=Number(d[i][1]); } } return ContentService.createTextOutput( JSON.stringify(r)).setMimeType( ContentService.MimeType.JSON); }
